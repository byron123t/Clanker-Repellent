/**
 * @brief Inclusive integer range check.
 * @param value The integer to test.
 * @param lower Lower bound of the range.
 * @param upper Upper bound of the range.
 * @return 1 if lower <= value <= upper, 0 otherwise.
 *         Returns 0 if lower > upper (reversed bounds).
 */
int in_range(int value, int lower, int upper) {
    if (lower > upper) {
        return 0;
    }
    if (value >= lower && value <= upper) {
        return 1;
    }
    return 0;
}
